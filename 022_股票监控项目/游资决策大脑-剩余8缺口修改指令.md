### 0.4 东财接口特性（**踩过的坑，必读**）
1. **clist/get 的 `po` 参数**：`po=1` = 降序（大在前），`po=0` = 升序（小在前）。`pz` 有上限约 100 条/页。**正负两端必须用 po=1 + po=0 双请求**（参考 `fetchBoardFundFlow` 的 all 模式 + `fetchBoardRankTopBottom`）。
2. **push2his.eastmoney.com**：对 nodejs `https.request` 做 TLS 指纹层 ban（12s socket hang up）。**不要用 proxy 转发 push2his**。如果需要分钟 K 线，要么浏览器直连（如果有 CORS），要么放弃用日级数据替代。
3. **emappdata.eastmoney.com（人气榜 POST）**：**完全支持 CORS**（OPTIONS 200 + Allow-Origin 回显任意 Origin），不校验 Referer/Origin → **浏览器直连即可，不要走 proxy**（proxy 反而失败）。
4. **dq.10jqka.com.cn（同花顺热度 GET）**：同样支持 CORS → 浏览器直连。
5. **proxy.js**：已改用 options 对象（hostname/path/servername）+ BROWSER_UA + Referer。白名单：push2*/datacenter-web/np-anotice-stock/np-weblist/emappdata/quote/qt.gtimg.cn/web.ifzq.gtimg.cn/stock.gtimg.cn/push2his/dq.10jqka.com.cn。新增白名单 host 在 `server/routes/proxy.js` 的 `ALLOWED_HOSTS` 数组加。
6. **ZBPool/DT 池 sort**：必须 `sort=fbt:asc`，其他 sort（fund:asc/lbc:desc）返回空数组（东财接口 bug）。

**步骤 2：新建极端板识别引擎**

```typescript
export type ExtremeType = "核按钮" | "地天板" | "天地板" | "无";
export interface ExtremeInput {
  code: string; name: string;
  yesterdayZt: boolean;     // 昨日是否涨停
  yesterdayHeight: number;  // 昨日连板数
  todayOpenPct: number | null;  // 今日开盘涨幅 %
  todayPct: number;         // 当前涨幅
  todayLowPct: number | null;   // 今日最低涨幅
}
export interface ExtremeVerdict {
  type: ExtremeType;
  confidence: number;
  reason: string;
}
export function detectExtremeBoard(input: ExtremeInput): ExtremeVerdict {
  // 核按钮：昨日涨停（尤其高位≥3板）+ 今日开盘 ≤-9% 或 当前 ≤-9%
  if (input.yesterdayZt && input.yesterdayHeight >= 2 &&
      ((input.todayOpenPct != null && input.todayOpenPct <= -9) || input.todayPct <= -9)) {
    return { type: "核按钮", confidence: 90, reason: `昨${input.yesterdayHeight}板今跌${input.todayPct.toFixed(1)}%，高位核按钮` };
  }
  // 地天板：今日最低 ≤-9.5%（触及跌停）+ 当前 ≥+9.5%（触及涨停）
  if (input.todayLowPct != null && input.todayLowPct <= -9.5 && input.todayPct >= 9.5) {
    return { type: "地天板", confidence: 85, reason: `最低${input.todayLowPct.toFixed(1)}%→现${input.todayPct.toFixed(1)}%，地天板弱转强` };
  }
  // 天地板：今日最高 ≥+9.5% + 当前 ≤-9.5%
  // （需要 todayHighPct，预留接口）
  return { type: "无", confidence: 0, reason: "" };
}
```

```typescript
export interface ContendInput {
  mainline: string;
  leaders: Array<{ code: string; name: string; height: number; sealFund?: number; fbt?: number; boardType?: string }>;
}
export interface ContendResult {
  status: "明确龙一" | "卡位胶着" | "梯队分散";
  leader?: string;  // 明确时
  contenders: string[];  // 卡位时
  reason: string;
}
export function detectLeaderContend(input: ContendInput): ContendResult {
  const sameHeight = input.leaders.filter(l => l.height === input.leaders[0]?.height);
  if (sameHeight.length <= 1) return { status: "明确龙一", leader: input.leaders[0]?.name, contenders: [], reason: "同高度唯一" };
  if (sameHeight.length >= 2 && sameHeight.length <= 3) {
    // 封单差距 <20% → 胶着
    const funds = sameHeight.map(l => l.sealFund ?? 0).sort((a,b) => b-a);
    const gap = funds[0] - (funds[1] ?? 0);
    if (gap < funds[0] * 0.2) return { status: "卡位胶着", contenders: sameHeight.map(l => l.name), reason: `同高度${sameHeight.length}只封单接近` };
  }
  return { status: "梯队分散", contenders: [], reason: "梯队分散" };
}
```

export function clusterAnnouncement(title: string, columnName: string): AnnCategory {
  if (/业绩|净利润|营收|预增|预减|扭亏|减亏/.test(title)) return "业绩";

export function detectAnnDensity(code: string, anns: Array<{stockCode: string; time: string}>): { density: number; alert: boolean } {
