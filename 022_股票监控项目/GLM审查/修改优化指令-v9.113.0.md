// 来源：/api/brain/context（market/limitLadder/mainlines/boardFund/sentiment）、/api/db/*（news/lhb/zt）
import { isLocalServer } from "./cloudStore";

export interface SourceMeta { source: "pg" | "push2" | "tencent" | "sina" | "push2delay" | "none"; asOf: number; stale: boolean; }

// 1) 结构化大盘快照（情绪/涨停/炸板/主线/资金/闸门）→ 永远 PG（cron 落库，新鲜且不被断）
export async function fetchMarketSnapshot(): Promise<{ data: any; meta: SourceMeta } | null> {
  if (!isLocalServer()) return null;
  try {
    const r = await fetch("/api/brain/context", { signal: AbortSignal.timeout(6000) });
    if (!r.ok) return null;
    const j = await r.json();
    const ts = j?.sources?.market ?? j?.snapshotVersion ?? Date.now();
    const stale = Date.now() - (ts ?? 0) > 25 * 60 * 1000; // >25min 标延迟

## 阶段 2：意图路由层（D-02/D-03）—— 简单问不再误进 ReAct

### T2-1 新建意图分类 `src/lib/intentRouter.ts`
```ts
// 三档路由：摘要(流式) / 数据(直读PG) / 决策(直达工具) / 复杂(ReAct精简)
export type Intent = "summary" | "data" | "decision" | "react" | "research";

export function classifyIntent(q: string): Intent {
  const t = q.trim();
  if (/个股深度调研/.test(t)) return "research";
  // 决策类（需准入/仓位/离场工具）
  if (/能不能上车|可不可以买|能不能买|仓位|止损|加仓|减仓|能不能上|要不要买|买不买/.test(t) || /\d{6}(怎么样|能不能|分析)/.test(t)) return "decision";
  // 纯数据类（直读 PG，不烧 LLM）
  if (/涨停(数|家数|多少)|跌停数|成交额|情绪(多少|分)|炸板率|溢价/.test(t)) return "data";
  // 纯摘要类（流式小 prompt）
  if (/一句话|总结|简述|概述|概况|速览|复盘|盘面|大盘|情绪如何|市场怎么样|今天(情况|行情)/.test(t)) return "summary";
  // 消息/主线/外围 → ReAct
  if (/消息|新闻|政策|主线|龙头|跟风|外盘|美股|纳指|隔夜|事件/.test(t)) return "react";
  return "summary"; // 默认走流式（最可靠）
}
```

### T2-2 AIConsole 按 intent 分流
- **文件**：`src/components/AIConsole.tsx`（`:138` 现 if isSimpleQuestion）。
- **改法**：用 `classifyIntent` 替代 `isSimpleQuestion`：
  - `summary` → `streamChat`（buildQuickSystem，小 prompt）
  - `data` → 直读 PG（`fetchMarketSnapshot`/`/api/db/*`），用模板组装带数字回答，**不调 LLM**（秒回、永不降级）
  - `decision` → `decisionDirect()`（见 T4-1）+ LLM 仅组织语言（或直接模板）
  - `react`/`research` → `runAssistantAgent`（ReAct，T3 精简 prompt）
- **验收**：
  - "一句话总结今日大盘"→ `summary` → 流式 AI 正文（不降级）✅（解决你当前痛点）
  - "今天涨停多少只"→ `data` → 秒回数字（PG，不烧 LLM）
  - "600519能不能上车"→ `decision` → 决策直达
  - "今天主线是什么哪些是龙头跟风"→ `react` → ReAct
  - 单测 `intentRouter.test.ts` 覆盖 5 档各 2 例。

1. **dataLayer PG-first**：PG 周期内为"最近 cron"，必须角标 asOf，勿冒充实时。

3. **ReAct prompt 瘦身**：工具子集别漏必备工具（如 getDecisionEvidence），否则 LLM 无法决策。
4. **length cap 12000**：增加配额消耗（仅 length 时），监控 OpenCode 用量。
5. **decisionDirect**：直接调纯函数须确认其无副作用依赖 LLM（若有，剥离纯逻辑）。
6. **改完必跑**：typegen→tsc→vitest→build→build_and_start，每阶段独立验证。
